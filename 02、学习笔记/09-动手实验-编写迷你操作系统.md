## 二、实验1：引导扇区（Boot Sector）

### 2.1 目标

实现一个512字节的引导扇区，在屏幕上打印"Hello OS!"

### 2.2 代码实现

```nasm
; boot.asm - 引导扇区程序
; 编译: nasm -f bin boot.asm -o boot.bin
; 运行: qemu-system-i386 -fda boot.bin

[BITS 16]           ; 16位实模式
[ORG 0x7C00]        ; BIOS将引导扇区加载到0x7C00

start:
    ; 设置段寄存器
    mov ax, 0
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7C00  ; 栈指针
    
    ; 清屏
    mov ah, 0x00
    mov al, 0x03    ; 80x25文本模式
    int 0x10
    
    ; 显示字符串
    mov si, message
    call print_string
    
    ; 死循环
    jmp $

; 打印字符串函数
print_string:
    mov ah, 0x0E    ; BIOS打印字符功能
.next:
    lodsb           ; 从SI加载字符到AL
    test al, al     ; 检查是否为0（字符串结束）
    jz .done
    int 0x10        ; 调用BIOS中断
    jmp .next
.done:
    ret

message db 'Hello OS! Welcome to Mini OS!', 0

; 填充至510字节
times 510-($-$$) db 0

; 魔数（引导扇区标识）
dw 0xAA55
```

### 2.3 编译运行

```bash
# 编译
nasm -f bin boot.asm -o boot.bin

# 运行
qemu-system-i386 -fda boot.bin

# 或使用dd写入软盘镜像
dd if=boot.bin of=floppy.img bs=512 count=1
qemu-system-i386 -fda floppy.img
```

### 2.4 知识点

```
1. BIOS启动过程：
   - 加电自检（POST）
   - 加载MBR（主引导记录）到0x7C00
   - 检查最后两字节是否为0xAA55
   - 跳转到0x7C00执行

2. 实模式：
   - 16位寻址
   - 最大1MB内存
   - 可直接访问硬件

3. BIOS中断：
   - INT 0x10: 视频服务
   - INT 0x13: 磁盘服务
   - INT 0x16: 键盘服务
```

## 三、实验2：进入保护模式

### 3.1 目标

从16位实模式切换到32位保护模式

### 3.2 代码实现

```nasm
; protected_mode.asm
[BITS 16]
[ORG 0x7C00]

start:
    ; 设置段寄存器
    mov ax, 0
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7C00
    
    ; 显示提示信息
    mov si, msg_real
    call print_string_16
    
    ; 加载GDT（全局描述符表）
    lgdt [gdt_descriptor]
    
    ; 开启保护模式（设置CR0的PE位）
    mov eax, cr0
    or eax, 1
    mov cr0, eax
    
    ; 远跳转到32位代码段
    jmp 0x08:protected_mode

; 16位打印函数
print_string_16:
    mov ah, 0x0E
.next:
    lodsb
    test al, al
    jz .done
    int 0x10
    jmp .next
.done:
    ret

msg_real db 'Entering protected mode...', 13, 10, 0

; GDT定义
gdt_start:
    ; 空描述符（必须）
    dq 0
    
    ; 代码段描述符
    gdt_code:
        dw 0xFFFF    ; 段界限低16位
        dw 0         ; 基址低16位
        db 0         ; 基址中8位
        db 10011010b ; 访问字节（代码段，可读）
        db 11001111b ; 标志+段界限高4位
        db 0         ; 基址高8位
    
    ; 数据段描述符
    gdt_data:
        dw 0xFFFF
        dw 0
        db 0
        db 10010010b ; 访问字节（数据段，可写）
        db 11001111b
        db 0

gdt_end:

gdt_descriptor:
    dw gdt_end - gdt_start - 1  ; GDT大小
    dd gdt_start                 ; GDT地址

[BITS 32]
; 32位保护模式代码
protected_mode:
    ; 设置数据段寄存器
    mov ax, 0x10      ; 数据段选择子
    mov ds, ax
    mov es, ax
    mov fs, ax
    mov gs, ax
    mov ss, ax
    mov esp, 0x90000  ; 设置32位栈指针
    
    ; 在显存中显示字符（保护模式下不能用BIOS中断）
    mov byte [0xB8000], 'P'
    mov byte [0xB8001], 0x0F  ; 白色字符
    mov byte [0xB8002], 'M'
    mov byte [0xB8003], 0x0F
    mov byte [0xB8004], 'O'
    mov byte [0xB8005], 0x0F
    mov byte [0xB8006], 'N'
    mov byte [0xB8007], 0x0F
    
    jmp $

times 510-($-$$) db 0
dw 0xAA55
```

### 3.3 知识点

```
保护模式特点：
1. 32位寻址，最大4GB内存
2. 分段和分页机制
3. 特权级（Ring 0-3）
4. 不能直接使用BIOS中断

GDT（全局描述符表）：
- 定义内存段的基址、界限、属性
- 每个描述符8字节
- 第一个描述符必须为空

切换过程：
1. 加载GDT（LGDT指令）
2. 设置CR0的PE位
3. 远跳转到32位代码段
```

## 四、实验3：进程调度模拟器

### 4.1 目标

用C语言实现进程调度算法模拟器

### 4.2 代码实现

```c
// process_scheduler.c
// 编译: gcc -o scheduler process_scheduler.c
// 运行: ./scheduler

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_PROCESSES 10

// 进程结构
typedef struct {
    int pid;
    int arrival_time;
    int burst_time;
    int priority;
    int remaining_time;
    int start_time;
    int finish_time;
    int waiting_time;
    int turnaround_time;
    int first_response;
    int completed;
} Process;

// 进程数组
Process processes[MAX_PROCESSES];
int num_processes = 0;

// 初始化进程
void init_process(int pid, int arrival, int burst, int priority) {
    processes[num_processes].pid = pid;
    processes[num_processes].arrival_time = arrival;
    processes[num_processes].burst_time = burst;
    processes[num_processes].priority = priority;
    processes[num_processes].remaining_time = burst;
    processes[num_processes].completed = 0;
    num_processes++;
}

// FCFS调度
void schedule_fcfs() {
    printf("\n=== FCFS调度 ===\n");
    int current_time = 0;
    
    printf("%-5s %-8s %-8s %-8s %-10s %-10s\n", 
           "PID", "到达", "服务", "完成", "周转", "等待");
    
    for (int i = 0; i < num_processes; i++) {
        if (current_time < processes[i].arrival_time) {
            current_time = processes[i].arrival_time;
        }
        
        processes[i].start_time = current_time;
        processes[i].finish_time = current_time + processes[i].burst_time;
        processes[i].turnaround_time = processes[i].finish_time - processes[i].arrival_time;
        processes[i].waiting_time = processes[i].turnaround_time - processes[i].burst_time;
        
        current_time = processes[i].finish_time;
        
        printf("%-5d %-8d %-8d %-8d %-10d %-10d\n",
               processes[i].pid, processes[i].arrival_time,
               processes[i].burst_time, processes[i].finish_time,
               processes[i].turnaround_time, processes[i].waiting_time);
    }
    
    // 计算平均值
    float avg_waiting = 0, avg_turnaround = 0;
    for (int i = 0; i < num_processes; i++) {
        avg_waiting += processes[i].waiting_time;
        avg_turnaround += processes[i].turnaround_time;
    }
    printf("\n平均等待时间: %.2f\n", avg_waiting / num_processes);
    printf("平均周转时间: %.2f\n", avg_turnaround / num_processes);
}

// SJF调度（非抢占）
void schedule_sjf() {
    printf("\n=== SJF调度 ===\n");
    int current_time = 0;
    int completed = 0;
    int order[MAX_PROCESSES];
    int order_idx = 0;
    
    Process temp[MAX_PROCESSES];
    memcpy(temp, processes, sizeof(Process));
    
    printf("%-5s %-8s %-8s %-8s %-10s %-10s\n", 
           "PID", "到达", "服务", "完成", "周转", "等待");
    
    while (completed < num_processes) {
        int min_burst = 9999;
        int min_idx = -1;
        
        // 找已到达且服务时间最短的进程
        for (int i = 0; i < num_processes; i++) {
            if (!temp[i].completed && temp[i].arrival_time <= current_time) {
                if (temp[i].burst_time < min_burst) {
                    min_burst = temp[i].burst_time;
                    min_idx = i;
                }
            }
        }
        
        if (min_idx == -1) {
            current_time++;
            continue;
        }
        
        temp[min_idx].start_time = current_time;
        temp[min_idx].finish_time = current_time + temp[min_idx].burst_time;
        temp[min_idx].turnaround_time = temp[min_idx].finish_time - temp[min_idx].arrival_time;
        temp[min_idx].waiting_time = temp[min_idx].turnaround_time - temp[min_idx].burst_time;
        temp[min_idx].completed = 1;
        
        current_time = temp[min_idx].finish_time;
        completed++;
        
        printf("%-5d %-8d %-8d %-8d %-10d %-10d\n",
               temp[min_idx].pid, temp[min_idx].arrival_time,
               temp[min_idx].burst_time, temp[min_idx].finish_time,
               temp[min_idx].turnaround_time, temp[min_idx].waiting_time);
    }
}

// 时间片轮转调度
void schedule_rr(int quantum) {
    printf("\n=== RR调度（时间片=%d）===\n", quantum);
    int current_time = 0;
    int completed = 0;
    
    Process temp[MAX_PROCESSES];
    memcpy(temp, processes, sizeof(Process));
    
    int queue[MAX_PROCESSES];
    int front = 0, rear = 0;
    
    // 初始将第一个进程入队
    queue[rear++] = 0;
    
    printf("执行顺序: ");
    
    while (completed < num_processes) {
        if (front == rear) {
            // 队列为空，找下一个到达的进程
            for (int i = 0; i < num_processes; i++) {
                if (!temp[i].completed && temp[i].arrival_time <= current_time) {
                    queue[rear++] = i;
                }
            }
            if (front == rear) {
                current_time++;
            }
            continue;
        }
        
        int idx = queue[front++];
        
        if (temp[idx].remaining_time <= quantum) {
            // 进程在当前时间片内完成
            current_time += temp[idx].remaining_time;
            temp[idx].remaining_time = 0;
            temp[idx].finish_time = current_time;
            temp[idx].completed = 1;
            temp[idx].turnaround_time = temp[idx].finish_time - temp[idx].arrival_time;
            temp[idx].waiting_time = temp[idx].turnaround_time - temp[idx].burst_time;
            completed++;
            printf("P%d(%d) ", temp[idx].pid, temp[idx].remaining_time + quantum);
        } else {
            // 时间片用完
            current_time += quantum;
            temp[idx].remaining_time -= quantum;
            queue[rear++] = idx;
            printf("P%d(%d) ", temp[idx].pid, quantum);
        }
        
        // 将新到达的进程入队
        for (int i = 0; i < num_processes; i++) {
            if (!temp[i].completed && temp[i].arrival_time <= current_time && 
                temp[i].remaining_time > 0) {
                int in_queue = 0;
                for (int j = front; j < rear; j++) {
                    if (queue[j] == i) {
                        in_queue = 1;
                        break;
                    }
                }
                if (!in_queue) {
                    queue[rear++] = i;
                }
            }
        }
    }
    printf("\n");
    
    printf("\n%-5s %-8s %-8s %-10s %-10s\n", "PID", "服务", "完成", "周转", "等待");
    for (int i = 0; i < num_processes; i++) {
        printf("%-5d %-8d %-8d %-10d %-10d\n",
               temp[i].pid, temp[i].burst_time, temp[i].finish_time,
               temp[i].turnaround_time, temp[i].waiting_time);
    }
}

int main() {
    // 初始化测试数据
    init_process(1, 0, 8, 3);
    init_process(2, 1, 4, 1);
    init_process(3, 2, 2, 4);
    init_process(4, 3, 1, 2);
    
    // 运行各调度算法
    schedule_fcfs();
    schedule_sjf();
    schedule_rr(2);
    
    return 0;
}
```

## 五、实验4：页面置换模拟器

### 5.1 目标

实现FIFO、LRU、OPT页面置换算法

### 5.2 代码实现

```c
// page_replacement.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_FRAMES 10
#define MAX_PAGES 100

int frames[MAX_FRAMES];
int num_frames;
int page_faults;

// 初始化
void init(int n) {
    num_frames = n;
    page_faults = 0;
    for (int i = 0; i < num_frames; i++) {
        frames[i] = -1;
    }
}

// 检查页是否在内存中
int is_in_memory(int page) {
    for (int i = 0; i < num_frames; i++) {
        if (frames[i] == page) return 1;
    }
    return 0;
}

// 找空位
int find_empty() {
    for (int i = 0; i < num_frames; i++) {
        if (frames[i] == -1) return i;
    }
    return -1;
}

// FIFO页面置换
int fifo(int pages[], int n) {
    init(3);
    int queue[MAX_FRAMES];
    int front = 0, rear = 0;
    
    printf("\n=== FIFO页面置换 ===\n");
    printf("页面序列: ");
    for (int i = 0; i < n; i++) printf("%d ", pages[i]);
    printf("\n\n");
    
    for (int i = 0; i < n; i++) {
        if (!is_in_memory(pages[i])) {
            page_faults++;
            
            int empty = find_empty();
            if (empty != -1) {
                frames[empty] = pages[i];
                queue[rear++] = empty;
            } else {
                int victim = queue[front++];
                frames[victim] = pages[i];
                queue[rear++] = victim;
            }
        }
        
        // 打印当前状态
        printf("访问 %d: [", pages[i]);
        for (int j = 0; j < num_frames; j++) {
            printf("%d ", frames[j]);
        }
        printf("] %s\n", is_in_memory(pages[i]) ? "" : "缺页");
    }
    
    printf("\n缺页次数: %d/%d = %.1f%%\n", 
           page_faults, n, (float)page_faults/n*100);
    return page_faults;
}

// LRU页面置换
int lru(int pages[], int n) {
    init(3);
    int last_used[MAX_FRAMES];
    for (int i = 0; i < num_frames; i++) last_used[i] = -1;
    int time = 0;
    
    printf("\n=== LRU页面置换 ===\n");
    
    for (int i = 0; i < n; i++) {
        int hit = 0;
        
        // 检查是否命中
        for (int j = 0; j < num_frames; j++) {
            if (frames[j] == pages[i]) {
                hit = 1;
                last_used[j] = time++;
                break;
            }
        }
        
        if (!hit) {
            page_faults++;
            
            int empty = find_empty();
            if (empty != -1) {
                frames[empty] = pages[i];
                last_used[empty] = time++;
            } else {
                // 找最久未使用的
                int min_time = last_used[0];
                int victim = 0;
                for (int j = 1; j < num_frames; j++) {
                    if (last_used[j] < min_time) {
                        min_time = last_used[j];
                        victim = j;
                    }
                }
                frames[victim] = pages[i];
                last_used[victim] = time++;
            }
        }
        
        printf("访问 %d: [", pages[i]);
        for (int j = 0; j < num_frames; j++) {
            printf("%d ", frames[j]);
        }
        printf("] %s\n", hit ? "命中" : "缺页");
    }
    
    printf("\n缺页次数: %d/%d = %.1f%%\n", 
           page_faults, n, (float)page_faults/n*100);
    return page_faults;
}

// OPT页面置换（最优）
int opt(int pages[], int n) {
    init(3);
    
    printf("\n=== OPT页面置换 ===\n");
    
    for (int i = 0; i < n; i++) {
        if (!is_in_memory(pages[i])) {
            page_faults++;
            
            int empty = find_empty();
            if (empty != -1) {
                frames[empty] = pages[i];
            } else {
                // 找将来最久不使用的
                int farthest = -1;
                int victim = 0;
                
                for (int j = 0; j < num_frames; j++) {
                    int next_use = n;  // 默认将来不用
                    
                    for (int k = i + 1; k < n; k++) {
                        if (pages[k] == frames[j]) {
                            next_use = k;
                            break;
                        }
                    }
                    
                    if (next_use > farthest) {
                        farthest = next_use;
                        victim = j;
                    }
                }
                
                frames[victim] = pages[i];
            }
        }
        
        printf("访问 %d: [", pages[i]);
        for (int j = 0; j < num_frames; j++) {
            printf("%d ", frames[j]);
        }
        printf("] %s\n", is_in_memory(pages[i]) ? "" : "缺页");
    }
    
    printf("\n缺页次数: %d/%d = %.1f%%\n", 
           page_faults, n, (float)page_faults/n*100);
    return page_faults;
}

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3, 0, 3, 2, 1, 2, 0, 1, 7, 0, 1};
    int n = sizeof(pages) / sizeof(pages[0]);
    
    printf("页面访问序列（共%d个）:\n", n);
    for (int i = 0; i < n; i++) {
        printf("%d ", pages[i]);
        if ((i + 1) % 10 == 0) printf("\n");
    }
    printf("\n\n");
    
    fifo(pages, n);
    lru(pages, n);
    opt(pages, n);
    
    printf("\n=== 总结 ===\n");
    printf("FIFO: 最优但无法实现（需预知未来）\n");
    printf("LRU: 接近最优，实际常用\n");
    printf("FIFO: 简单但可能产生Belady异常\n");
    
    return 0;
}
```

## 六、实验5：磁盘调度模拟器

### 6.1 目标

实现FCFS、SSTF、SCAN、C-SCAN磁盘调度算法

### 6.2 代码实现

```c
// disk_scheduler.c
#include <stdio.h>
#include <stdlib.h>

#define MAX_REQUESTS 100

int requests[MAX_REQUESTS];
int num_requests;
int head_position;
int max_track;

// 比较函数（用于排序）
int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

// FCFS
int fcfs() {
    int total = 0;
    int current = head_position;
    
    printf("\n=== FCFS磁盘调度 ===\n");
    printf("服务顺序: ");
    
    for (int i = 0; i < num_requests; i++) {
        total += abs(current - requests[i]);
        current = requests[i];
        printf("%d ", current);
    }
    printf("\n总寻道距离: %d\n", total);
    return total;
}

// SSTF
int sstf() {
    int total = 0;
    int current = head_position;
    int visited[MAX_REQUESTS] = {0};
    
    printf("\n=== SSTF磁盘调度 ===\n");
    printf("服务顺序: ");
    
    for (int count = 0; count < num_requests; count++) {
        int min_dist = 9999;
        int min_idx = -1;
        
        for (int i = 0; i < num_requests; i++) {
            if (!visited[i]) {
                int dist = abs(current - requests[i]);
                if (dist < min_dist) {
                    min_dist = dist;
                    min_idx = i;
                }
            }
        }
        
        visited[min_idx] = 1;
        total += min_dist;
        current = requests[min_idx];
        printf("%d ", current);
    }
    printf("\n总寻道距离: %d\n", total);
    return total;
}

// SCAN（电梯算法）
int scan(int direction) {
    int sorted[MAX_REQUESTS];
    for (int i = 0; i < num_requests; i++) sorted[i] = requests[i];
    qsort(sorted, num_requests, sizeof(int), compare);
    
    int total = 0;
    int current = head_position;
    
    printf("\n=== SCAN磁盘调度（方向=%s）===\n", 
           direction == 1 ? "向外" : "向内");
    printf("服务顺序: ");
    
    if (direction == 1) {  // 向外
        for (int i = 0; i < num_requests; i++) {
            if (sorted[i] >= head_position) {
                total += abs(current - sorted[i]);
                current = sorted[i];
                printf("%d ", current);
            }
        }
        // 到达边界
        total += abs(current - max_track);
        current = max_track;
        
        // 反向向内
        for (int i = num_requests - 1; i >= 0; i--) {
            if (sorted[i] < head_position) {
                total += abs(current - sorted[i]);
                current = sorted[i];
                printf("%d ", current);
            }
        }
    } else {  // 向内
        for (int i = num_requests - 1; i >= 0; i--) {
            if (sorted[i] <= head_position) {
                total += abs(current - sorted[i]);
                current = sorted[i];
                printf("%d ", current);
            }
        }
        total += current;
        current = 0;
        
        for (int i = 0; i < num_requests; i++) {
            if (sorted[i] > head_position) {
                total += abs(current - sorted[i]);
                current = sorted[i];
                printf("%d ", current);
            }
        }
    }
    printf("\n总寻道距离: %d\n", total);
    return total;
}

// C-SCAN
int cscan() {
    int sorted[MAX_REQUESTS];
    for (int i = 0; i < num_requests; i++) sorted[i] = requests[i];
    qsort(sorted, num_requests, sizeof(int), compare);

        if (sorted[i] >= head_position) {

        }
    }
    
    // 回到起点

    total += max_track;

    // 继续向外

            printf("%d ", current);
        }
    }
    printf("\n总寻道距离: %d\n", total);

}
```

## 七、实验6：银行家算法模拟器

### 7.1 目标

实现银行家算法进行死锁避免

### 7.2 代码实现

```c
// banker_algorithm.c
#include <stdio.h>
#include <stdlib.h>

#define MAX_PROCESSES 10
#define MAX_RESOURCES 10

int available[MAX_RESOURCES];
int max_claim[MAX_PROCESSES][MAX_RESOURCES];
int allocation[MAX_PROCESSES][MAX_RESOURCES];
int need[MAX_PROCESSES][MAX_RESOURCES];
int num_processes, num_resources;

// 初始化
void init() {
    num_processes = 5;
    num_resources = 3;
    
    // 可用资源
    available[0] = 3; available[1] = 3; available[2] = 2;
    
    // 最大需求
    int max[5][3] = {{7,5,3}, {3,2,2}, {9,0,2}, {2,2,2}, {4,3,3}};
    for (int i = 0; i < num_processes; i++)
        for (int j = 0; j < num_resources; j++)
            max_claim[i][j] = max[i][j];
    
    // 已分配
    int alloc[5][3] = {{0,1,0}, {2,0,0}, {3,0,2}, {2,1,1}, {0,0,2}};
    for (int i = 0; i < num_processes; i++)
        for (int j = 0; j < num_resources; j++)
            allocation[i][j] = alloc[i][j];
    
    // 计算Need
    for (int i = 0; i < num_processes; i++)
        for (int j = 0; j < num_resources; j++)
            need[i][j] = max_claim[i][j] - allocation[i][j];
}

// 打印状态
void print_state() {
    printf("\n%-8s %-15s %-15s %-15s\n", "进程", "Max", "Allocation", "Need");
    for (int i = 0; i < num_processes; i++) {
        printf("P%-7d ", i);
        printf("(%d,%d,%d)    ", max_claim[i][0], max_claim[i][1], max_claim[i][2]);
        printf("(%d,%d,%d)    ", allocation[i][0], allocation[i][1], allocation[i][2]);
        printf("(%d,%d,%d)\n", need[i][0], need[i][1], need[i][2]);
    }
    printf("\nAvailable: (%d,%d,%d)\n", available[0], available[1], available[2]);
}

// 安全性检查
int is_safe() {
    int work[MAX_RESOURCES];
    int finish[MAX_PROCESSES] = {0};
    int safe_seq[MAX_PROCESSES];
    int count = 0;
    
    for (int j = 0; j < num_resources; j++) work[j] = available[j];
    
    while (count < num_processes) {
        int found = 0;
        
        for (int i = 0; i < num_processes; i++) {
            if (!finish[i]) {
                // 检查Need[i] <= Work
                int can_finish = 1;
                for (int j = 0; j < num_resources; j++) {
                    if (need[i][j] > work[j]) {
                        can_finish = 0;
                        break;
                    }
                }
                
                if (can_finish) {
                    for (int j = 0; j < num_resources; j++)
                        work[j] += allocation[i][j];
                    safe_seq[count++] = i;
                    finish[i] = 1;
                    found = 1;
                }
            }
        }
        
        if (!found) break;
    }
    
    if (count == num_processes) {
        printf("安全序列: ");
        for (int i = 0; i < count; i++) printf("P%d ", safe_seq[i]);
        printf("\n");
        return 1;
    }
    return 0;
}

// 银行家算法
int banker(int process_id, int request[]) {
    printf("\n进程P%d请求: (%d,%d,%d)\n", process_id, request[0], request[1], request[2]);
    
    // 1. 检查Request <= Need
    for (int j = 0; j < num_resources; j++) {
        if (request[j] > need[process_id][j]) {
            printf("错误：请求超过最大需求\n");
            return 0;
        }
    }
    
    // 2. 检查Request <= Available
    for (int j = 0; j < num_resources; j++) {
        if (request[j] > available[j]) {
            printf("资源不足，等待\n");
            return 0;
        }
    }
    
    // 3. 试探分配
    for (int j = 0; j < num_resources; j++) {
        available[j] -= request[j];
        allocation[process_id][j] += request[j];
        need[process_id][j] -= request[j];
    }
    
    // 4. 安全性检查
    if (is_safe()) {
        printf("可以分配！\n");
        return 1;
    } else {
        // 回滚
        for (int j = 0; j < num_resources; j++) {
            available[j] += request[j];
            allocation[process_id][j] -= request[j];
            need[process_id][j] += request[j];
        }
        printf("不安全，拒绝分配\n");
        return 0;
    }
}

int main() {
    init();
    
    printf("=== 银行家算法模拟器 ===\n");
    print_state();
    
    printf("\n--- 初始安全性检查 ---\n");
    if (is_safe()) {
        printf("系统处于安全状态\n");
    }
    
    // 测试请求
    printf("\n--- 测试1: P1请求(1,0,2) ---\n");
    int req1[] = {1, 0, 2};
    banker(1, req1);
    
    printf("\n--- 测试2: P4请求(3,3,0) ---\n");
    int req2[] = {3, 3, 0};
    banker(4, req2);
    
    printf("\n--- 测试3: P0请求(0,2,0) ---\n");
    int req3[] = {0, 2, 0};
    banker(0, req3);
    
    return 0;
}
```
