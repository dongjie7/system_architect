## 一、I/O系统概述

### 1.1 I/O设备分类

| 类型 | 特点 | 示例 |
|------|------|------|
| **字符设备** | 按字符为单位传输 | 键盘、打印机、串口 |
| **块设备** | 按数据块为单位传输 | 磁盘、SSD、光盘 |
| **网络设备** | 按包/帧传输 | 网卡 |

### 1.2 I/O控制方式演进

```
程序直接控制 → 中断驱动 → DMA → 通道控制

CPU参与度：高 → 低
效率：低 → 高
复杂度：低 → 高
```

## 二、I/O控制方式（核心考点）

### 2.1 程序直接控制方式

```c
// 程序直接控制I/O（轮询）
void program_controlled_io() {
    // 忙等待（Busy Waiting）
    while ((status_register & READY_BIT) == 0) {
        // CPU一直循环等待，浪费CPU时间
    }
    
    // 设备准备好，读取数据
    data = data_register;
    
    // 继续处理...
}

/*
优点：简单
缺点：CPU利用率极低（忙等待）
适用：早期简单系统
*/
```

### 2.2 中断驱动方式

```c
// 中断驱动I/O
void interrupt_driven_io() {
    // 1. 向设备发送读命令
    device_command = READ;
    
    // 2. CPU继续执行其他程序
    do_other_work();
    
    // 3. 设备完成后发出中断
    // 4. CPU响应中断，执行中断处理程序
}

// 中断处理程序
void io_interrupt_handler() {
    // 保存现场
    save_context();
    
    // 读取数据
    data = data_register;
    
    // 处理数据
    process_data(data);
    
    // 恢复现场
    restore_context();
}

/*
优点：CPU可以并行做其他事
缺点：每次数据传输都需要中断，中断开销大
适用：中低速设备
*/
```

### 2.3 DMA（直接存储器访问）方式

```c
// DMA 控制I/O
void dma_io() {
    // 1. CPU初始化DMA控制器
    dma_controller.base_address = buffer_address;  // 内存缓冲区地址
    dma_controller.count = data_size;              // 传输数据量
    dma_controller.direction = READ;               // 传输方向
    dma_controller.device = DISK;                  // 目标设备
    
    // 2. 启动DMA传输
    dma_controller.start = 1;
    
    // 3. CPU继续执行其他程序
    do_other_work();
    
    // 4. DMA完成一个数据块后，发出中断
    // 5. CPU处理DMA完成中断
}

// DMA完成中断处理
void dma_completion_handler() {
    // 检查是否还有数据要传输
    if (remaining_data > 0) {
        // 更新地址和计数，继续传输
        dma_controller.base_address += block_size;
        dma_controller.count -= block_size;
        dma_controller.start = 1;
    } else {
        // 全部传输完成
        notify_process();
    }
}

/*
优点：数据块直接在设备和内存间传输，无需CPU参与
缺点：需要DMA控制器硬件支持
适用：高速块设备（磁盘、网卡）
*/
```

### 2.4 通道控制方式

```c
// 通道控制I/O
void channel_io() {
    // 1. CPU构建通道程序（Channel Program）
    ChannelProgram program;
    program.commands[0] = {READ, disk_address, buffer1, size1};
    program.commands[1] = {READ, disk_address + size1, buffer2, size2};
    program.commands[2] = {READ, disk_address + size1 + size2, buffer3, size3};
    program.end = true;
    
    // 2. CPU向通道发出I/O指令
    channel.start(&program);
    
    // 3. CPU继续执行
    do_other_work();
    
    // 4. 通道完成整个程序后发出中断
}

/*
优点：通道可执行通道程序，管理多个设备
缺点：硬件复杂
适用：大型计算机系统
*/
```

### 2.5 四种I/O控制方式对比

| 方式 | CPU参与度 | 数据传输单位 | 适用设备 | 硬件要求 |
|------|-----------|--------------|----------|----------|
| 程序直接控制 | 全程参与 | 字符 | 低速 | 无 |
| 中断驱动 | 每字符中断 | 字符 | 中低速 | 中断控制器 |
| DMA | 块传输中断 | 数据块 | 高速块设备 | DMA控制器 |
| 通道 | 程序完成中断 | 多块 | 多设备系统 | 通道控制器 |

## 三、DMA详解

### 3.1 DMA工作原理

```
DMA传输过程：
1. CPU初始化DMA控制器（设置地址、长度、方向）
2. CPU启动DMA传输
3. DMA控制器接管总线，直接在设备和内存间传输数据
4. 每传输一个数据块，DMA发出中断
5. CPU处理中断（或DMA完成全部传输后中断）
6. DMA释放总线，CPU继续执行

DMA与CPU的关系：
- 周期性挪用（Cycle Stealing）：DMA每传输一个字就归还总线
- 停止CPU访问：DMA传输期间CPU暂停访问内存
```

### 3.2 DMA代码实现

```c
// DMA控制器结构
struct DMAController {
    unsigned int base_address;    // 内存基址
    unsigned int count;           // 传输字节数
    unsigned int device_address;  // 设备地址
    unsigned int direction;       // 0=读设备，1=写设备
    unsigned int status;          // 状态：0=空闲，1=忙，2=完成
    unsigned int interrupt;       // 中断标志
};

// DMA传输模拟
void dma_transfer(DMAController *dma, void *buffer, int device_addr, int size, int direction) {
    // 初始化
    dma->base_address = (unsigned int)buffer;
    dma->device_address = device_addr;
    dma->count = size;
    dma->direction = direction;
    dma->status = 1;  // 忙
    
    // 模拟DMA传输
    unsigned char *mem_ptr = (unsigned char *)buffer;
    
    for (int i = 0; i < size; i++) {
        if (direction == 0) {
            // 从设备读
            mem_ptr[i] = read_device(dma->device_address + i);
        } else {
            // 写到设备
            write_device(dma->device_address + i, mem_ptr[i]);
        }
    }
    
    // 传输完成
    dma->status = 2;  // 完成
    dma->interrupt = 1;  // 触发中断
}

// DMA中断处理
void dma_interrupt_service(DMAController *dma) {
    if (dma->interrupt) {
        dma->interrupt = 0;
        dma->status = 0;  // 空闲
        
        // 通知等待进程
        wakeup_waiting_process(dma);
    }
}
```

## 四、设备分配

### 4.1 设备分配数据结构

```c
// 系统设备表（SDT）
struct SystemDeviceTable {
    int device_type;        // 设备类型
    int device_id;          // 设备编号
    int device_status;      // 设备状态
    int device_address;     // 设备地址
    DCT *dct;               // 指向设备控制表
};

// 设备控制表（DCT）
struct DCT {
    int device_id;          // 设备编号
    int device_status;      // 忙/闲
    int device_type;        // 设备类型
    COCT *coct;             // 指向控制器控制表
    Queue wait_queue;       // 设备等待队列
};

// 控制器控制表（COCT）
struct COCT {
    int controller_id;      // 控制器编号
    int controller_status;  // 忙/闲
    int channel_id;         // 连接的通道
    Queue wait_queue;       // 控制器等待队列
};

// 通道控制表（CHCT）
struct CHCT {
    int channel_id;         // 通道编号
    int channel_status;     // 忙/闲
    Queue wait_queue;       // 通道等待队列
};
```

### 4.2 设备分配算法

```c
// 设备分配过程
int allocate_device(int device_type, int process_id) {
    // 1. 查找空闲设备
    DCT *dct = find_free_device(device_type);
    if (dct == NULL) {
        // 无空闲设备，加入等待队列
        add_to_wait_queue(device_type, process_id);
        return -1;
    }
    
    // 2. 分配设备
    dct->device_status = BUSY;
    dct->process_id = process_id;
    
    // 3. 分配控制器
    COCT *coct = dct->coct;
    if (coct->controller_status == BUSY) {
        // 控制器忙，等待
        add_to_controller_wait(coct, process_id);
        return -1;
    }
    coct->controller_status = BUSY;
    
    // 4. 分配通道
    CHCT *chct = coct->chct;
    if (chct->channel_status == BUSY) {
        // 通道忙，等待
        add_to_channel_wait(chct, process_id);
        return -1;
    }
    chct->channel_status = BUSY;
    
    // 5. 建立通路
    establish_path(chct, coct, dct);
    
    return device_id;
}

// 释放设备
void release_device(int device_id) {
    DCT *dct = get_dct(device_id);
    
    // 释放设备
    dct->device_status = FREE;
    
    // 释放控制器
    COCT *coct = dct->coct;
    coct->controller_status = FREE;
    
    // 释放通道
    CHCT *chct = coct->chct;
    chct->channel_status = FREE;
    
    // 唤醒等待进程
    if (!is_empty(dct->wait_queue)) {
        wakeup_first(dct->wait_queue);
    }
}
```

### 4.3 设备分配考虑因素

| 因素 | 说明 |
|------|------|
| 设备固有特性 | 设备类型、速度、是否可共享 |
| 进程需求 | 进程优先级、I/O时间要求 |
| 系统状态 | 设备/控制器/通道状态 |
| 分配算法 | 先来先服务、优先级等 |

## 五、SPOOLing技术

### 5.1 SPOOLing概念

**SPOOLing**（Simultaneous Peripheral Operations On-Line）= 假脱机技术

```
目的：将独占设备改造为共享设备

原理：
1. 利用磁盘模拟独占设备
2. 输入/输出在磁盘上进行
3. 实际I/O由SPOOLing守护进程完成

组成：
- 输入井/输出井（磁盘空间）
- 输入缓冲区/输出缓冲区（内存）
- SPOOLing守护进程
```

### 5.2 SPOOLing系统结构

```
用户进程              SPOOLing系统              独占设备
   │                      │                        │
   │  请求打印            │                        │
   ├─────────────────────→│                        │
   │                      │  写入输出井             │
   │                      ├─────────────────────→  │
   │  返回（假完成）       │                        │
   │←─────────────────────│                        │
   │                      │                        │
   │                      │  SPOOLing守护进程       │
   │                      │  从输出井读取           │
   │                      │←─────────────────────  │
   │                      │  实际打印               │
   │                      ├───────────────────────→│
```

### 5.3 SPOOLing代码实现

```c
// SPOOLing系统结构
struct SpoolingSystem {
    // 输出井（磁盘）
    char output_well[MAX_WELL_SIZE];
    int well_head;
    int well_tail;
    
    // 输出缓冲区（内存）
    char output_buffer[BUFFER_SIZE];
    int buffer_count;
    
    // 请求队列
    Queue print_queue;
    
    // 守护进程
    pthread_t daemon_thread;
};

// 用户进程请求打印
void spool_print_request(SpoolingSystem *spool, char *data, int size, int process_id) {
    // 1. 创建打印请求
    PrintRequest *req = malloc(sizeof(PrintRequest));
    req->process_id = process_id;
    req->data = strdup(data);
    req->size = size;
    
    // 2. 写入输出井
    memcpy(spool->output_well + spool->well_tail, data, size);
    spool->well_tail = (spool->well_tail + size) % MAX_WELL_SIZE;
    req->well_offset = spool->well_tail - size;
    
    // 3. 加入请求队列
    enqueue(spool->print_queue, req);
    
    // 4. 立即返回（假完成）
    printf("打印请求已接受\n");
}

// SPOOLing守护进程
void *spool_daemon(void *arg) {
    SpoolingSystem *spool = (SpoolingSystem *)arg;
    
    while (1) {
        // 等待打印请求
        if (is_empty(spool->print_queue)) {
            sleep(1);
            continue;
        }
        
        // 取出请求
        PrintRequest *req = dequeue(spool->print_queue);
        
        // 从输出井读取数据到缓冲区
        memcpy(spool->output_buffer, 
               spool->output_well + req->well_offset, 
               req->size);
        
        // 实际输出到设备
        printer_write(spool->output_buffer, req->size);
        
        // 释放请求
        free(req->data);
        free(req);
    }
}
```

## 六、缓冲管理

### 6.1 缓冲技术

| 类型 | 说明 | 适用场景 |
|------|------|----------|
| **单缓冲** | 一个缓冲区 | 简单场景 |
| **双缓冲** | 两个缓冲区交替 | 生产者-消费者 |
| **循环缓冲** | 多个缓冲区组成环 | 数据流 |
| **缓冲池** | 多个缓冲区统一管理 | 现代OS |

### 6.2 单缓冲与双缓冲

```c
// 单缓冲
void single_buffer() {
    char buffer[BUFFER_SIZE];
    
    // 生产者
    while (1) {
        produce_data(buffer);      // 写入缓冲区
        while (buffer_full) {};    // 等待消费者取走
        buffer_full = 1;
    }
    
    // 消费者
    while (1) {
        while (!buffer_full) {};   // 等待生产者写入
        consume_data(buffer);      // 从缓冲区读取
        buffer_full = 0;
    }
}

// 双缓冲
void double_buffer() {
    char buffer1[BUFFER_SIZE];
    char buffer2[BUFFER_SIZE];
    char *in_buffer = buffer1;
    char *out_buffer = buffer2;
    
    // 生产者写入in_buffer
    // 同时消费者读取out_buffer
    // 完成后交换指针
    swap(in_buffer, out_buffer);
}
```
